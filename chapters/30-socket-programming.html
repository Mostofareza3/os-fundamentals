HTTP আসলে কিছু লেখা byte

HTTP/1.1 একটা text protocol: TCP-র byte-এর স্রোতের উপর কয়েকটা লাইন, প্রতিটা \r\n-এ শেষ। প্রথম লাইনে method আর path, তারপর header, তারপর একটা খালি লাইন, তারপর body। Framework ছাড়া, শুধু net দিয়ে একটা HTTP server:

EXPERIMENT — mini-http.js
// mini-http.js — express বা http module ছাড়া, শুধু net দিয়ে একটা HTTP server
const net = require('net');

net.createServer(sock => {
  let buf = '';
  sock.on('data', chunk => {
    buf += chunk;
    const end = buf.indexOf('\r\n\r\n');        // HTTP header শেষ হয় একটা খালি লাইনে
    if (end === -1) return;                       // পুরো header এখনো আসেনি (Chapter 27-এর framing)
    const [requestLine, ...headers] = buf.slice(0, end).split('\r\n');
    console.log('request line:', JSON.stringify(requestLine));
    console.log('headers     :', headers.length, 'lines, e.g.', JSON.stringify(headers[0]));
    const body = JSON.stringify({ hello: 'from raw TCP', path: requestLine.split(' ')[1] }) + '\n';
    sock.end(
      'HTTP/1.1 200 OK\r\n' +
      'Content-Type: application/json\r\n' +
      `Content-Length: ${Buffer.byteLength(body)}\r\n` +   // body কোথায় শেষ — framing
      'Connection: close\r\n' +
      '\r\n' + body
    );
  });
}).listen(8080, '127.0.0.1', () => console.log('listening on 8080'));

এক terminal-এ node mini-http.js, আরেকটায় curl -s -i http://127.0.0.1:8080/users/42:

HTTP/1.1 200 OK
Content-Type: application/json
Content-Length: 44
Connection: close

{"hello":"from raw TCP","path":"/users/42"}

Server-এর দিকে: request line: "GET /users/42 HTTP/1.1" আর headers : 3 lines, e.g. "Host: 127.0.0.1:8080"। এমনকি request নিজে হাতে লিখেও পাঠানো যায় — nc (netcat) শুধু stdin-এর byte একটা TCP connection-এ ঢেলে দেয়:

printf 'GET /by-hand HTTP/1.1\r\nHost: me\r\n\r\n' | nc 127.0.0.1 8080

উত্তর আসে {"hello":"from raw TCP","path":"/by-hand"} সহ একই header।

curl সন্তুষ্ট, কারণ HTTP-র চাহিদা এটুকুই: ঠিক ক্রমে ঠিক byte। বাকি সব — URL-এর query ভাঙা, header-এর case, chunked body, keep-alive, Content-Length না মিললে কী করা — Node-এর http module-এর কাজ, আর তার উপরে Express-এর routing। কিন্তু নিচে যা আছে তা এই: একটা socket, কিছু byte, আর framing।

চেনা error গুলোর আসল মানে

Network error-এর নামগুলো kernel-এর দেওয়া, আর প্রতিটা আগের chapter-গুলোর কোনো একটা ঘটনার নাম। পাঁচটা নিজেরাই ঘটাই:

EXPERIMENT (Linux / Docker) — errors.js
// errors.js — চেনা network error গুলো, একটা একটা করে ঘটিয়ে
const net = require('net');
const once = (emitter, ev) => new Promise(r => emitter.once(ev, r));
const log = (name, e) => console.log(name.padEnd(28), e ? e.code : 'no error');

(async () => {
  // ১) একই port দুইবার
  const a = net.createServer().listen(8080, '127.0.0.1');
  await once(a, 'listening');
  const b = net.createServer().listen(8080, '127.0.0.1');
  log('listen twice on 8080:', await once(b, 'error'));

  // ২) কেউ শুনছে না এমন port
  log('connect to closed port:', await once(net.connect(8081, '127.0.0.1'), 'error'));

  // ৩) server connection "ছিঁড়ে" দেয় (RST), client পড়ছিল
  const rst = net.createServer(s => s.resetAndDestroy()).listen(8082, '127.0.0.1');
  await once(rst, 'listening');
  log('peer sends RST:', await once(net.connect(8082, '127.0.0.1'), 'error'));

  // ৪) ওপাশ বন্ধ হয়ে যাওয়ার পরেও লিখে যাওয়া
  const bye = net.createServer(s => s.destroy()).listen(8083, '127.0.0.1');
  await once(bye, 'listening');
  const c = net.connect({ port: 8083, host: '127.0.0.1', allowHalfOpen: true });
  c.on('connect', () => { const t = setInterval(() => c.write('x'), 10); c.on('close', () => clearInterval(t)); });
  log('write after peer closed:', await once(c, 'error'));

  // ৫) কোনো উত্তরই আসে না (packet কোথাও হারিয়ে যায়)
  const t0 = Date.now();
  const e = await once(net.connect(81, '10.255.255.1'), 'error');
  log(`nobody answers (${((Date.now() - t0) / 1000).toFixed(1)} s):`, e);
  process.exit(0);
})();

connect to closed port:      ECONNREFUSED
peer sends RST:              ECONNRESET
write after peer closed:     EPIPE
nobody answers (3.1 s):      ETIMEDOUT
  • EADDRINUSE — bind() ব্যর্থ: port-টা অন্য কোনো socket-এর দখলে। প্রায়ই আগের process এখনো চলছে (lsof -iTCP:8080 দিয়ে খোঁজো)।
  • ECONNREFUSED — তোমার SYN-এর জবাবে ওপাশের kernel RST পাঠিয়েছে: "এই port-এ কেউ শুনছে না।" মানে machine পর্যন্ত পৌঁছেছ, program নেই।
  • ECONNRESET — চালু connection-এ ওপাশ থেকে RST এসেছে: ওপাশের process crash করেছে, kill -9 হয়েছে, বা কোনো load balancer/proxy idle connection কেটে দিয়েছে। তোমার code-এর bug হওয়া জরুরি না — ওপাশের দিকে খোঁজো।
  • EPIPE — এমন connection-এ লিখছ যেটা ওপাশ ইতিমধ্যে বন্ধ করেছে। প্রথম লেখার জবাবে ওপাশ RST দেয়; তার পরের লেখায় kernel বলে "পাইপ ভাঙা"।
  • ETIMEDOUT — SYN পাঠিয়ে কোনো জবাবই আসেনি (firewall নীরবে ফেলে দিচ্ছে, বা ঠিকানায় কোনো machine নেই); kernel নির্দিষ্ট কয়েকবার আবার পাঠিয়ে হাল ছেড়েছে। কতবার, সেটা net.ipv4.tcp_syn_retries — Docker VM-এ default ৬, যাতে প্রায় ২ মিনিট লাগে; তাই পরীক্ষায় container-এর জন্য ১ করে দিয়েছি, ৩.১ সেকেন্ডে শেষ। Production-এ connect-এ নিজের timeout দাও; ২ মিনিট অপেক্ষা কারও কাজে লাগে না।

fd-র সীমা: EMFILE

প্রতিটা socket একটা fd, আর প্রতিটা process-এর খোলা fd-র একটা সীমা আছে — ulimit -n। Server-এর দিকে প্রতিটা connection একটা fd, client-এর দিকেও একটা। সীমা ছুঁলে কী হয়, ইচ্ছা করে ছোট সীমা বসিয়ে দেখি:

EXPERIMENT (Linux / Docker) — emfile.js
// emfile.js — প্রতিটা connection একটা fd; fd-র সীমা ছুঁলে কী হয়
const net = require('net');
const fs = require('fs');
console.log('fds already open at start:', fs.readdirSync('/proc/self/fd').length);
for (let i = 1; i <= 100; i++) {
  net.connect(8080, '127.0.0.1').on('error', e => {
    console.log(`connection #${i}: ${e.code}`);
    process.exit(0);
  });
}
docker run --rm -v "$PWD":/w:ro node:22-bookworm-slim sh -c '
  node -e "require(\"net\").createServer(() => {}).listen(8080)" & sleep 0.5
  ulimit -n 64; node /w/emfile.js'
fds already open at start: 18
connection #47: EMFILE

Node নিজেই শুরুতে ১৭টা fd খোলা রাখে (stdin/stdout/stderr, epoll, eventfd, pipe…; ১৮-র একটা readdirSync-এর নিজের), তাই ৬৪-র সীমায় ৪৬টা connection-এর পরে ৪৭তমটায় EMFILE — "too many open files"। Docker-এ default সীমা 1048576, কিন্তু সব জায়গায় এত না: আমার Mac-এ launchctl limit maxfiles দেখায় soft সীমা ২৫৬। Production-এ EMFILE প্রায়ই আসে fd leak থেকে: বন্ধ না করা socket (Chapter 26-এর CLOSE-WAIT), বন্ধ না করা file। ls /proc/<pid>/fd | wc -l সময়ের সাথে শুধু বাড়তে থাকলে leak খোঁজো।

Graceful shutdown: ভদ্রভাবে বন্ধ

Deploy-এর সময় পুরনো process বন্ধ হয়। হঠাৎ process.exit() করলে চলমান request গুলো মাঝপথে কাটা পড়ে — client পায় ECONNRESET। ভদ্র উপায় (graceful shutdown): নতুন connection নেওয়া বন্ধ করো, চলমানগুলো শেষ হতে দাও, তারপর বেরোও। server.close() ঠিক এটাই করে — listening socket বন্ধ করে, কিন্তু accepted socket গুলো ছোঁয় না:

EXPERIMENT — graceful.js
// graceful.js — server.close(): নতুন দরজা বন্ধ, কিন্তু চলমান কাজ শেষ হতে দাও
const net = require('net');
const t0 = Date.now();
const log = msg => console.log(`${String(Date.now() - t0).padStart(5)} ms  ${msg}`);

const server = net.createServer(sock => {
  log('server: request in, working for 1 s...');
  setTimeout(() => sock.end('done\n'), 1000);      // একটা ধীর request
});
server.listen(8080, '127.0.0.1', () => {
  const a = net.connect(8080, '127.0.0.1');
  a.on('data', d => log(`client A got: ${d.toString().trim()}`));

  setTimeout(() => {
    log('SIGTERM-এর মতো: server.close()');
    server.close(() => log('server: all connections finished, "close" event'));
    net.connect(8080, '127.0.0.1').on('error', e => log(`client B (new): ${e.code}`));
  }, 200);
});
    5 ms  server: request in, working for 1 s...
  207 ms  SIGTERM-এর মতো: server.close()
  210 ms  client B (new): ECONNREFUSED
 1009 ms  client A got: done
 1010 ms  server: all connections finished, "close" event

close()-এর ৩ ms পরে নতুন client B ECONNREFUSED — দরজা বন্ধ। কিন্তু আগে থেকে চলা A-র request পুরো সময় নিয়ে শেষ হয়েছে, আর তারপরেই server-এর 'close' event। বাস্তবে এটা বসে SIGTERM signal-এর handler-এ: Kubernetes pod বন্ধ করার সময় আগে SIGTERM পাঠায়, আর default ৩০ সেকেন্ড পরে SIGKILL। তাই process.on('SIGTERM', () => server.close(() => process.exit(0)))। একটা ফাঁদ: HTTP keep-alive-এর অলস connection গুলোও "চলমান", তাই close() তাদের জন্যও অপেক্ষা করে — http server-এ server.closeIdleConnections() সেগুলো ছেড়ে দেয়।

COMMON MISCONCEPTION

"ECONNRESET এসেছে মানে আমার code-এ bug।" ECONNRESET মানে ওপাশের kernel একটা RST পাঠিয়েছে — errors.js-এ server নিজে resetAndDestroy() করে সেটা ঘটিয়েছে। বাস্তবে উৎস প্রায়ই ওপাশে: ওপাশের process crash বা kill, বা মাঝের load balancer idle connection কেটে দেওয়া। Error-এর নামটা kernel-এর ঘটনার নাম — কোন ঘটনা, সেটা বুঝলেই খোঁজার দিক ঠিক হয়।

MENTAL MODEL

Node-এর socket API kernel-এর system call-এর পাতলা মোড়ক: listen = socket+bind+listen, 'connection' = accept, write/'data' = write/read, end = FIN, resetAndDestroy = RST। HTTP এর উপরে কিছু লেখা byte আর framing। প্রতিটা network error একটা kernel-ঘটনার নাম — কে কাকে RST পাঠাল, কে জবাবই দিল না, কার fd ফুরাল। আর server বন্ধ মানে আগে দরজা বন্ধ (close), তারপর ভেতরের কাজ শেষ হওয়ার অপেক্ষা।
